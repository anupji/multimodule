public class MinDeletionsToMakeStringBalanced {
    
    public int solution(String s) {
        int openCount = 0;
        int deletions = 0;

        for (char c : s.toCharArray()) {
            if (c == '(') {
                openCount++;
            } else if (c == ')') {
                if (openCount > 0) {
                    openCount--;
                } else {
                    deletions++;
                }
            }
        }

        // Remaining opening brackets need to be deleted
        deletions += openCount;

        return deletions;
    }

    public static void main(String[] args) {
        String s = "))((";
        System.out.println("Minimum deletions required: " + minDeletionsToBalance(s));
    }
}

public class FormatString {
    
    public static String obtainFormatString(String s) {
        int countA = 0, countB = 0;
        
        for (char c : s.toCharArray()) {
            if (c == 'A') {
                countA++;
            } else if (c == 'B') {
                countB++;
            }
        }
        
        StringBuilder result = new StringBuilder();
        
        if (countA > countB) {
            for (int i = 0; i < countB; i++) {
                result.append('A');
            }
            for (int i = 0; i < countB; i++) {
                result.append('B');
            }
        } else {
            for (int i = 0; i < countA; i++) {
                result.append('A');
            }
        }

        return result.toString();
    }

    public static void main(String[] args) {
        String s = "AABBBAAAB"; // Example input string
        String formattedString = obtainFormatString(s);
        System.out.println("Formatted string: " + formattedString);
    }
}






public class MaximumBananaMove {
    
    public static int maxBananaMove(String s) {
        // Count occurrences of each letter
        int[] count = new int[26];
        for (char c : s.toCharArray()) {
            count[c - 'A']++;
        }

        // Calculate how many times "BANANA" can be formed
        int maxMoves = Integer.MAX_VALUE;
        maxMoves = Math.min(maxMoves, count['B' - 'A']); // 'B' appears only once in "BANANA"
        maxMoves = Math.min(maxMoves, count['A' - 'A'] / 3); // 'A' appears three times in "BANANA"
        maxMoves = Math.min(maxMoves, count['N' - 'A'] / 2); // 'N' appears twice in "BANANA"
        
        return maxMoves;
    }

    public static void main(String[] args) {
        String s = "BANANAS";
        System.out.println("Maximum number of times the move can be applied: " + maxBananaMove(s));
    }
}






public class StringOperations {
    
    public String solution(String s, String t) {
        int n = s.length();
        int m = t.length();

        if (n == m) {
            if (s.equals(t)) {
                return "EQUAL";
            } else if (isInsert(s, t)) {
                return "INSERT";
            } else if (isRemove(s, t)) {
                return "REMOVE";
            } else if (isSwap(s, t)) {
                return "SWAP";
            }
        } else if (n + 1 == m) {
            if (isInsert(s, t)) {
                return "INSERT";
            }
        } else if (n - 1 == m) {
            if (isRemove(s, t)) {
                return "REMOVE";
            }
        }

        return "IMPOSSIBLE";
    }

    private boolean isInsert(String s, String t) {
        int i = 0, j = 0;

        while (i < s.length() && j < t.length()) {
            if (s.charAt(i) != t.charAt(j)) {
                if (i != j) return false;
                j++;
            } else {
                i++;
                j++;
            }
        }

        return true;
    }

    private  boolean isRemove(String s, String t) {
        return isInsert(t, s);
    }

    private  boolean isSwap(String s, String t) {
        int firstDiff = -1, secondDiff = -1;

        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) != t.charAt(i)) {
                if (firstDiff == -1) {
                    firstDiff = i;
                } else if (secondDiff == -1) {
                    secondDiff = i;
                } else {
                    return false;
                }
            }
        }

        return (firstDiff != -1 && secondDiff != -1 && 
                s.charAt(firstDiff) == t.charAt(secondDiff) && 
                s.charAt(secondDiff) == t.charAt(firstDiff));
    }

    public static void main(String[] args) {
        String s = "abcd";
        String t1 = "abdc"; // SWAP
        String t2 = "abcde"; // INSERT
        String t3 = "abc"; // REMOVE
        String t4 = "abcdf"; // IMPOSSIBLE

        System.out.println(stringOperation(s, t1)); // SWAP
        System.out.println(stringOperation(s, t2)); // INSERT
        System.out.println(stringOperation(s, t3)); // REMOVE
        System.out.println(stringOperation(s, t4)); // IMPOSSIBLE
    }
}

public class MethodReferenceExample {
    public static void main(String[] args) {
        // Reference to a static method
        Function<Integer, String> staticMethodRef = MethodReferenceExample::staticMethod;
        System.out.println(staticMethodRef.apply(5)); // Output: The number is 5

        // Reference to an instance method of a particular object
        MethodReferenceExample instance = new MethodReferenceExample();
        Function<String, String> instanceMethodRef = instance::instanceMethod;
        System.out.println(instanceMethodRef.apply("world")); // Output: Hello, world!

        // Reference to an instance method of an arbitrary object of a particular type
        Function<String, Integer> lengthMethodRef = String::length;
        System.out.println(lengthMethodRef.apply("Java")); // Output: 4

        // Reference to a constructor
        Supplier<MethodReferenceExample> constructorRef = MethodReferenceExample::new;
        MethodReferenceExample newObj = constructorRef.get();
        newObj.method(); // Output: Constructor called
    }

    // Static method
    public static String staticMethod(Integer num) {
        return "The number is " + num;
    }

    // Instance method
    public String instanceMethod(String name) {
        return "Hello, " + name + "!";
    }

    // Constructor
    public MethodReferenceExample() {
        System.out.println("Constructor called");
    }

    // Another method
    public void method() {
        System.out.println("Method called");
    }
}

import java.util.stream.IntStream;

public class IntStreamExample {
    public static void main(String[] args) {
        // Creating an IntStream from a range
        IntStream.range(1, 10)
                 .forEach(System.out::println); // Prints numbers from 1 to 9

        // Creating an IntStream from an array
        int[] numbers = {1, 2, 3, 4, 5};
        IntStream streamFromArray = IntStream.of(numbers);
        streamFromArray.forEach(System.out::println); // Prints numbers 1 to 5

        // Using map to transform elements
        IntStream.range(1, 5)
                 .map(x -> x * 2)
                 .forEach(System.out::println); // Prints even numbers from 2 to 8

        // Using filter to select elements
        IntStream.range(1, 10)
                 .filter(x -> x % 2 == 0)
                 .forEach(System.out::println); // Prints even numbers from 2 to 8

        // Using sum to calculate the sum of elements
        int sum = IntStream.range(1, 6)
                          .sum();
        System.out.println("Sum: " + sum); // Prints "Sum: 15"
    }
}


import org.springframework.security.authentication.AuthenticationProvider;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.AuthenticationException;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.stereotype.Component;

@Component
public class CustomAuthenticationProvider implements AuthenticationProvider {

    private final UserDetailsService userDetailsService;

    public CustomAuthenticationProvider(UserDetailsService userDetailsService) {
        this.userDetailsService = userDetailsService;
    }

    @Override
    public Authentication authenticate(Authentication authentication) throws AuthenticationException {
        String username = authentication.getName();
        String password = authentication.getCredentials().toString();

        UserDetails userDetails = userDetailsService.loadUserByUsername(username);

        // Perform custom authentication logic here
        // For example, compare the provided password with the one stored in the UserDetails object
        if (password.equals(userDetails.getPassword())) {
            return new UsernamePasswordAuthenticationToken(username, password, userDetails.getAuthorities());
        } else {
            throw new BadCredentialsException("Invalid username or password");
        }
    }

    @Override
    public boolean supports(Class<?> authentication) {
        return authentication.equals(UsernamePasswordAuthenticationToken.class);
    }
}

import org.springframework.context.annotation.Configuration;
import org.springframework.security.authentication.AuthenticationProvider;
import org.springframework.security.config.annotation.authentication.builders.AuthenticationManagerBuilder;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.annotation.web.configuration.WebSecurityConfigurerAdapter;

@Configuration
@EnableWebSecurity
public class SecurityConfig extends WebSecurityConfigurerAdapter {

    private final AuthenticationProvider authenticationProvider;

    public SecurityConfig(AuthenticationProvider authenticationProvider) {
        this.authenticationProvider = authenticationProvider;
    }

    @Override
    protected void configure(AuthenticationManagerBuilder auth) throws Exception {
        auth.authenticationProvider(authenticationProvider);
    }
}
